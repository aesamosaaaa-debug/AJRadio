    private fun addStation(name: String, url: String) {
        val s = St(name.replace(Regex("[\t\n]"), " "), url, false)
        all.add(s); save(); pick(s); refresh()
    }

    private fun searchDialog() {
        val q = EditText(this).apply { hint = "اسم المحطة أو البلد"; imeOptions = EditorInfo.IME_ACTION_SEARCH; setSingleLine() }
        val res = mutableListOf<Pair<String, String>>()
        val ad = ArrayAdapter(this, android.R.layout.simple_list_item_1, mutableListOf<String>())
        val list = ListView(this).apply { adapter = ad }
        val box = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(dp(16), dp(8), dp(16), 0)
            addView(q); addView(list, LinearLayout.LayoutParams(-1, dp(320)))
        }
        val dlg = AlertDialog.Builder(this).setTitle("بحث عن محطة").setView(box).setNegativeButton("إغلاق", null).create()
        q.setOnEditorActionListener { _, _, _ ->
            val t = q.text.toString().trim()
            if (t.isNotEmpty()) Thread {
                val r = Cfg.search(t)
                runOnUiThread {
                    res.clear(); res.addAll(r); ad.clear(); ad.addAll(r.map { it.first })
                    if (r.isEmpty()) toast("لا نتائج أو لا يوجد اتصال")
                }
            }.start()
            true
        }
        list.setOnItemClickListener { _, _, i, _ ->
            addStation(res[i].first, res[i].second); toast("أُضيفت المحطة"); dlg.dismiss()
        }
        dlg.show()
    }

    private fun buildUi() {
        window.statusBarColor = BG
        window.navigationBarColor = BG
        val root = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setPadding(dp(16), dp(36), dp(16), dp(28))
            layoutDirection = View.LAYOUT_DIRECTION_RTL

        fun tv(t: String, sz: Float, c: Int = TXT, bold: Boolean = false) = TextView(this).apply {

        }
        fun lab(t: String) = tv(t, 13f, MUTE).apply { setPadding(0, 0, 0, dp(8)) }
        fun card() = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL; background = rr(CARD, 22)

            layoutParams = LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(12) }

            if (pos != sleepPos) {

        }
        root.addView(player)
        root.addView(section("المحطات", spin, favOnly, row))

        root.addView(section("المنبّه بالإذاعة", alTime, alSw))
        setContentView(ScrollView(this).apply { setBackgroundColor(BG); addView(root) })

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        prefs = getSharedPreferences("aj", MODE_PRIVATE)
        load()
        if (Build.VERSION.SDK_INT >= 33) requestPermissions(arrayOf(Manifest.permission.POST_NOTIFICATIONS), 1)
        buildUi()
    }

    override fun onStart() {
        super.onStart()
        if (prefs.getBoolean("alOn", false)) Cfg.setAlarm(this)
        val f = MediaController.Builder(this, SessionToken(this, ComponentName(this, PlaybackService::class.java))).buildAsync()
        fut = f
        f.addListener({
            try { ctl = f.get(); ctl?.addListener(lis); updateUi() } catch (e: Exception) { status.text = "تعذّر الاتصال بالخدمة" }
        }, ContextCompat.getMainExecutor(this))
    }

    override fun onStop() {
        fut?.let { MediaController.releaseFuture(it) }
        fut = null; ctl = null
        super.onStop()
    }
