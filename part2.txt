    private fun load() {
        all.clear()
        val raw = prefs.getString("st", null) ?: "${Cfg.DEF_NAME}\t${Cfg.DEF_URL}\t0"
        raw.split("\n").forEach {
            val p = it.split("\t")
            if (p.size >= 3) all.add(St(p[0], p[1], p[2] == "1"))
        }
    }

    private fun save() {
        prefs.edit().putString("st", all.joinToString("\n") { "${it.name}\t${it.url}\t${if (it.fav) 1 else 0}" }).apply()

        val sel = prefs.getString("sel", "")
        vis = all.filter { !favOnly.isChecked || it.fav }
        spin.items(vis.map { (if (it.fav) "★ " else "") + it.name })
        val i = vis.indexOfFirst { it.url == sel }
        if (i >= 0) spin.setSelection(i)
    }

    private fun cur(): St? = vis.getOrNull(spin.selectedItemPosition)

    private fun playSelected() {
        val c = ctl ?: return
        val s = cur() ?: return
        err = null
        c.setMediaItem(Cfg.item(s.url, s.name, prefs.getInt("buf", 1)))
        c.prepare()
        c.play()
    }

    private fun toggle() {
        val c = ctl ?: return
        val s = cur() ?: return
        if (c.playWhenReady && c.playbackState != Player.STATE_IDLE) {
            c.pause()
        } else if (c.currentMediaItem?.localConfiguration?.uri?.toString() == s.url &&
            c.playbackState != Player.STATE_IDLE
        ) {
            err = null; c.seekToDefaultPosition(); c.play()
        } else playSelected()
    }

        val c = ctl

            on -> "جارٍ التحميل…"
            err != null -> err
            else -> "متوقف"
        }
    }
