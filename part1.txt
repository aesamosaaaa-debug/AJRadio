package com.example.ajradio

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import androidx.media3.common.MediaItem
import androidx.media3.common.MediaMetadata
import org.json.JSONArray
import java.net.HttpURLConnection
import java.net.URL
import java.net.URLEncoder
import java.util.Calendar

object Cfg {
    const val DEF_URL = "https://live-hls-audio-web-aja.getaj.net/VOICE-AJA/01.m3u8"
    const val DEF_NAME = "الجزيرة"
    val sec = intArrayOf(10, 30, 90)
    val off = longArrayOf(6000, 20000, 45000)

    fun live(idx: Int) = MediaItem.LiveConfiguration.Builder().setTargetOffsetMs(off[idx]).build()

    fun item(url: String, name: String, idx: Int): MediaItem = MediaItem.Builder()
        .setUri(url)
        .setMediaMetadata(MediaMetadata.Builder().setTitle(name).setArtist("راديو الانترنت").build())
        .setLiveConfiguration(live(idx))
        .build()

    fun saved(c: Context): MediaItem {
        val p = c.getSharedPreferences("aj", 0)
        return item(p.getString("sel", DEF_URL)!!, p.getString("selName", DEF_NAME)!!, p.getInt("buf", 1))
    }

    fun op(c: Context, action: String, code: Int): PendingIntent = PendingIntent.getBroadcast(
        c, code, Intent(c, ActionReceiver::class.java).setAction(action),
        PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
    )

    fun setAlarm(c: Context) {
        val p = c.getSharedPreferences("aj", 0)
        val cal = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, p.getInt("alH", 7)); set(Calendar.MINUTE, p.getInt("alM", 0))
            set(Calendar.SECOND, 0); set(Calendar.MILLISECOND, 0)
            if (timeInMillis <= System.currentTimeMillis() + 1000) add(Calendar.DAY_OF_YEAR, 1)
        }
        val show = PendingIntent.getActivity(c, 1, Intent(c, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE)
        c.getSystemService(AlarmManager::class.java)
            .setAlarmClock(AlarmManager.AlarmClockInfo(cal.timeInMillis, show), op(c, "aj.ALARM", 2))
    }

    fun cancelAlarm(c: Context) {
        c.getSystemService(AlarmManager::class.java).cancel(op(c, "aj.ALARM", 2))
    }

    fun search(q: String): List<Pair<String, String>> {
        for (h in listOf("de1", "nl1", "fi1")) {
            try {
                val u = URL("https://${h}.api.radio-browser.info/json/stations/search?limit=30&hidebroken=true&order=clickcount&reverse=true&name=" + URLEncoder.encode(q, "UTF-8"))
                val con = u.openConnection() as HttpURLConnection
                con.setRequestProperty("User-Agent", "InternetRadio/2.0")
                con.connectTimeout = 8000; con.readTimeout = 8000
                val a = JSONArray(con.inputStream.bufferedReader().readText())
                return (0 until a.length()).map {
                    val o = a.getJSONObject(it)
                    val cc = o.optString("countrycode")
                    Pair(o.optString("name").trim() + (if (cc.isEmpty()) "" else " ($cc)"), o.optString("url_resolved"))
                }.filter { it.second.startsWith("http") }
            } catch (e: Exception) { }
        }
        return emptyList()
    }
}

package com.example.ajradio

import android.appwidget.AppWidgetManager
import android.appwidget.AppWidgetProvider
import android.content.BroadcastReceiver
import android.content.ComponentName
import android.content.Context
import android.content.Intent
import android.widget.RemoteViews
import androidx.core.content.ContextCompat
import androidx.media3.session.MediaController
import androidx.media3.session.SessionToken

class ActionReceiver : BroadcastReceiver() {
    override fun onReceive(c: Context, i: Intent) {
        val pr = goAsync()
        val action = i.action
        if (action == "aj.ALARM" && c.getSharedPreferences("aj", 0).getBoolean("alOn", false)) Cfg.setAlarm(c)
        val f = MediaController.Builder(c, SessionToken(c, ComponentName(c, PlaybackService::class.java))).buildAsync()
        f.addListener({
            try {
                val m = f.get()
                if (action == "aj.TOGGLE" && m.playWhenReady) m.pause()
                else { m.setMediaItem(Cfg.saved(c)); m.prepare(); m.play() }
            } catch (e: Exception) { }
            pr.finish()
        }, ContextCompat.getMainExecutor(c))
    }
}

class RadioWidget : AppWidgetProvider() {
    override fun onUpdate(c: Context, m: AppWidgetManager, ids: IntArray) {
        val v = RemoteViews(c.packageName, R.layout.widget)
        v.setOnClickPendingIntent(R.id.wroot, Cfg.op(c, "aj.TOGGLE", 3))
        ids.forEach { m.updateAppWidget(it, v) }
